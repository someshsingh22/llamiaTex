\block{Motivation}{
\keyline{How can an LLM effectively collaborate with\\
non-language agents?}
\begin{itemize}
\item LLMs have become general purpose orchestrators that can solve complex domain specific tasks by coordinating with tools and domain expert subagents.
\item However, in many important domains such as robotics and game playing, the strongest available agents are \textbf{not language models}. Their expertise lives in rich internal representations (e.g.\ AlphaZero, RT-1).
\item We study this collaboration using \textbf{chess} as a controlled environment owing to strong open engines (Lc0), public datasets, and real world tasks.
\end{itemize}
}

  \end{itemize}

\block{Our Approach}{
  \begin{itemize}
    \item A \textbf{language agent} (Qwen3 LLM) collaborates with a
      frozen \textbf{non-language agent} (Lc0-BT4 chess engine) and can update its state dynamically through verbalized \action{actions}.
    \item \textbf{Latent state internalization:} the engine's internal
      state is directly projected and concatenated into the LLM's Chain of Thought as
      \state{state tokens}, instead of being verbalized.
    \item The LLM is trained with \textbf{reinforcement learning} to
      reason over \lang{language}, \action{action} and \state{state}
      tokens. We call the resulting model \textbf{\llamia}.
  \end{itemize}
}

\block{Latent State Internalization}{
  \begin{center}
    \includegraphics[width=\linewidth]{plots/llamia-fig (12).pdf}
  \end{center}
  \figcap{Illustration of internalized CoT. The LLM reasons in \lang{language},
    calls the engine on the current board state, plays an \action{action} that
    changes the board. The LatentBridge projects Lc0 (subagent)'s internal board representations or \state{states} into the LLM's CoT at each invokation}
  \vspace{0.6em}
  \begin{itemize}
    \item \textbf{Projection.} Lc0's hidden state
      $\mathbf{h}_s$ (layer 14 of 15) is mapped by a
      3-layer MLP, the \textbf{LatentBridge}, to $k{=}32$ \state{state
      tokens} in the LLM's embedding space:
      $\;\mathbf{z}_s = H_\varphi(\mathbf{h}_s)$.
    \item \textbf{Dynamic invocation.} Inside its chain of thought, the LLM can repeatedly decide \textbf{when} to call the subagent and execute \action{actions} that update the board to counterfactual states, and call \texttt{get\_policy} on any position. Each call
      re-encodes the current state, so the LLM always reasons over the
      position it is analyzing.
  \end{itemize}
}

\block{Training with Reinforcement Learning}{
  \begin{minipage}[c]{0.6\linewidth}
    \begin{itemize}
      \item \textbf{Stage 1: projector alignment.} We follow LLaVA style two stage training, training only the LatentBridge on large scale synthetic state--value pairs.
      \item \textbf{Stage 2: reinforcement learning.} We post-train the LLM and
        LatentBridge jointly with \textbf{DAPO} on six of LLAMIA-Bench tasks. \state{State tokens} are masked from the
        policy gradient. The engine stays frozen in both stages.
      \item \textbf{Verbalization Baseline:} We also train \llamiaverb{} replacing the state tokens with verbalized summaries (best moves, evaluation, position)
    \end{itemize}
  \end{minipage}\hfill
  \begin{minipage}[c]{0.38\linewidth}
    \centering
    \includegraphics[width=\linewidth]{plots/poster_training_reward.pdf}
  \end{minipage}
  \par\vspace{0.2em}
  {\large\bfseries\color{slate} Emergent collaboration strategies}\par\vspace{0.1em}
  \begin{minipage}[c]{0.44\linewidth}
    \begin{itemize}
    \item We annotate the LLM and Subagent collaboration patterns using GPT-4o e.g. Follow, where the LLM simply follows the engine.
      \item \llamia{} \textbf{adapts its strategy to the task}: engine-follow for
        gameplay, consult-then-override for behavior cloning,
        counterfactual queries for commentary.
      \item \llamiaverb{} \textbf{collapses to engine-follow} on every task (62--76\%).
    \end{itemize}
