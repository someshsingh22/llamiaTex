\subsection{GRPO Training Dynamics}
\label{sec:results_grpo}

\begin{figure*}[t]
    \centering
    \includegraphics[width=\linewidth]{pages/figures/llamia_grpo_curves.png}
    \caption{\textbf{GRPO training reward: internalization vs.\ verbalized tool use.}
    Mean reward vs.\ training step at three model sizes (4B, 8B, 27B).
    Solid lines: LLAMIA (latent-state internalization).
    Dashed lines: Lc0 as an external tool call (LLAMIA-Verb).
    Shaded bands: running standard deviation.
    Internalization consistently reaches $2$--$3\times$ higher reward than
    verbalized tool use across all sizes, with the gap widening during training.
    }
    \label{fig:grpo_curves}
\end{figure*}

\begin{figure*}[t]
    \centering
    \includegraphics[width=\linewidth]{pages/figures/task_reward_curves.png}
    \caption{\textbf{Per-task GRPO reward curves (LLAMIA vs.\ LLAMIA-Verb, 14B).}
    LLAMIA (blue, solid) optimizes reward across all seven tasks.
    LLAMIA-Verb (orange, dashed) achieves partial signal on Behavior
    Cloning and Difficulty but stays flat on Interest, Game Commentary,
    and Instruction Planning---the tasks where the target signal is
    either non-verbalizable (Interest) or requires multi-step state
    tracking (Commentary, Planning). The Interest panel shows
    LLAMIA-Verb's reward hacking: it learns the statistical prior that
    minimizes MAE without genuinely reading the aesthetic signal.
    }
    \label{fig:task_reward_curves}
\end{figure*}

\subsection{LLAMIA-Bench Overview}
\label{sec:results_bench}

\begin{figure}[t]
    \centering
    \includegraphics[width=\linewidth]{pages/figures/llamia_bench_radar.png}
    \caption{\textbf{LLAMIA-Bench radar: four systems across seven tasks.}
    LLAMIA (dark blue) dominates all spokes.
    Frontier+Tool (orange) collapses on Interest (non-verbalizable
    signal) and Instruction Planning (requires policy override).
    LLAMIA-Verb (light blue) tracks Frontier+Tool on single-position
    tasks but degrades further on multi-step tasks due to
    narrative fragmentation from RL-trained tool-call patterns.
    Task Experts (green) are strong on their training distributions
    but collapse on wild splits and tasks they were not designed for.
    }
    \label{fig:radar}
\end{figure}

\begin{figure*}[t]
    \centering
    \includegraphics[width=\linewidth]{pages/figures/llamia_bench_bars.png}
    \caption{\textbf{LLAMIA-Bench grouped bar chart.}
    Performance (normalized) across all seven tasks.
    Numbers above LLAMIA bars show absolute scores.
    Key patterns: (1)~LLAMIA is uniformly high across all tasks;
    (2)~Interest is the diagnostic task where all non-LLAMIA systems
    collapse ($<$25 vs.\ LLAMIA's 82);
    (3)~Task Experts win on in-distribution BC (MAIA) but lose on
    wild splits;
    (4)~LLAMIA-Verb consistently underperforms Frontier+Tool on
    Commentary and Instruction Planning.
    }
    \label{fig:bench_bars}
\end{figure*}

\subsection{Verbalization Gap Decomposition}
\label{sec:results_vg}

\begin{figure}[t]
    \centering
    \includegraphics[width=\linewidth]{pages/figures/vg_decomposition.png}
    \caption{\textbf{Two mechanisms behind the verbalization gap.}
    \textit{Left:} depth-compounding---the gap between LLAMIA and
    LLAMIA-Verb grows monotonically as tasks require more sustained
    interaction (Gameplay $\to$ Rationale $\to$ Commentary $\to$ ICP).
    \textit{Right:} non-verbalizability---on the same Puzzle
    Understanding positions, the verbalization gap on Interest is
    $4.4\times$ larger than on Difficulty, demonstrating that some
    agent signals have no text representation regardless of
    interaction depth.
    }
    \label{fig:vg_decomposition}
\end{figure}
